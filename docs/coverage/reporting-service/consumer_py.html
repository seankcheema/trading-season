1""" 

2The reporting consumer process: consumer group reporting-ingester. 

3 

4Runs as its own container (python consumer.py) beside the gunicorn web 

5service. It polls trade-events and platform-records, appends each message to 

6that topic's event store, commits the offset only after the line is on disk, 

7and runs the report job on a schedule. Exactly one instance may run at a 

8time: two processes must never append to the same partition file. 

9 

10These two topics are everything the reporting service knows. Neither this 

11process nor the web service connects to the trading database. 

12"""

27class KafkaSource: 

28 """Thin wrapper over confluent_kafka.Consumer so the loop can be tested with a fake."""

30 def __init__(self, bootstrap_servers, group_id, topics, on_assign=None): 

31 from confluent_kafka import Consumer # imported here so tests need no broker library 

32 

33 topics = [topics] if isinstance(topics, str) else list(topics) 

34 self._consumer = Consumer({

53class TopicStores: 

54 """ 

55 The event store of each consumed topic, so a message lands in its own 

56 topic's files. stores maps the name the broker knows a topic by to the 

57 store of what it carries. 

58 """ 

59 

60 def __init__(self, stores): 

61 self._stores = dict(stores) 

62 

63 def refresh(self): 

64 for store in self._stores.values(): 

65 store.refresh() 

66 

67 def append(self, message, key, body): 

68 """Append to the store of the message's topic. A topic nobody stores is not written.""" 

69 store = self._stores.get(message.topic()) 

70 if store is None: 

71 logger.warning('Skipping message from unexpected topic %s', message.topic()) 

72 return False 

73 return store.append(message.partition(), message.offset(), key, body) 

74

76def handle_message(message, store): 

77 """ 

78 Append one message to the store. Returns True when a line was written, 

79 False when it was a duplicate or unreadable. Both outcomes commit. store 

80 is one topic's EventStore, or the TopicStores of every consumed topic. 

81 """ 

82 key = message.key() 

83 value = message.value() 

84 key = key.decode('utf-8') if isinstance(key, bytes) else key 

85 value = value.decode('utf-8') if isinstance(value, bytes) else value 

86 try: 

87 body = json.loads(value) if value else None 

88 except json.JSONDecodeError: 

89 body = None 

90 if not isinstance(body, dict): 

91 logger.warning('Skipping unreadable message partition=%s offset=%s', 

92 message.partition(), message.offset()) 

93 return False 

94 if isinstance(store, TopicStores): 

95 written = store.append(message, key, body) 

96 else: 

97 written = store.append(message.partition(), message.offset(), key, body) 

98 if written and body.get('type') is not None: 

99 logger.info('Stored platform record type=%s key=%s partition=%s offset=%s', 

100 body.get('type'), key, message.partition(), message.offset()) 

101 elif written: 

102 logger.info('Stored trade event orderId=%s status=%s key=%s partition=%s offset=%s', 

103 body.get('orderId'), body.get('status'), key, message.partition(), message.offset()) 

104 return written

107def run_consumer_loop(source, store, stop_event, poll_timeout=1.0): 

108 """ 

109 Poll until stop_event is set. Every message is handled then committed 

110 synchronously, so nothing is pending to commit on shutdown. 

111 """ 

112 stats = {'written': 0, 'skipped': 0, 'errors': 0} 

113 try: 

114 while not stop_event.is_set(): 

115 message = source.poll(poll_timeout) 

116 if message is None: 

117 continue 

118 if message.error(): 

119 stats['errors'] += 1 

120 logger.error('Kafka error: %s', message.error()) 

121 continue 

122 if handle_message(message, store): 

123 stats['written'] += 1 

124 else: 

125 stats['skipped'] += 1 

126 source.commit(message) 

127 finally: 

128 source.close() 

129 logger.info('Consumer stopped: %s', stats) 

130 return stats

133def start_scheduler(app, files_dir, interval_minutes, first_run_at=None): 

134 """ 

135 Run the report job on an interval inside the Flask app context. The first 

136 run is one interval away unless first_run_at names an earlier time. 

137 

138 The job brings this process's read model up to date first: a run takes 

139 its trade data from the event files and the names and trader levels it 

140 labels them with from the read model built out of those same files. 

141 """ 

142 from models import db 

143 from read_model import read_model 

144 from report_run import run_report 

145 

146 def job(): 

147 with app.app_context(), read_model.lock: 

148 try: 

149 read_model.sync(db.session, files_dir) 

150 run_report(files_dir, db.session) 

151 except Exception: # noqa: BLE001 - the job must never kill the scheduler 

152 logger.exception('Report run failed') 

153 finally: 

154 db.session.remove() 

155 

156 # Passed only when set: APScheduler reads next_run_time=None as a paused job. 

157 first_run = {'next_run_time': first_run_at} if first_run_at is not None else {} 

158 # No misfire grace limit: a run that comes due while the process is busy, 

159 # as the first one does during start-up, still runs instead of being dropped. 

160 scheduler = BackgroundScheduler() 

161 scheduler.add_job(job, trigger='interval', minutes=interval_minutes, id='report_run', 

162 coalesce=True, max_instances=1, misfire_grace_time=None, **first_run) 

163 scheduler.start() 

164 logger.info('Report run scheduled every %s minutes', interval_minutes) 

165 return scheduler

168def main(): 

169 logging.basicConfig(level=logging.INFO, format='%(asctime)s %(levelname)s %(name)s %(message)s') 

170 from app import app 

171 

172 config = app.config 

173 files_dir = config['REPORTING_FILES_DIR'] 

174 events_dir = f'{files_dir}/events' 

175 stores = TopicStores({ 

176 config['KAFKA_TRADE_EVENTS_TOPIC']: EventStore(events_dir, TRADE_EVENTS), 

177 config['KAFKA_PLATFORM_RECORDS_TOPIC']: EventStore(events_dir, PLATFORM_RECORDS), 

178 }) 

179 topics = list(stores._stores) 

180 

181 stop = threading.Event() 

182 for signum in (signal.SIGTERM, signal.SIGINT): 

183 signal.signal(signum, lambda *_: stop.set()) 

184 

185 # The first run is due at once, so a restarted consumer serves a report of 

186 # the events already on disk instead of nothing for a whole interval. 

187 scheduler = None 

188 if config['SCHEDULER_ENABLED']: 

189 scheduler = start_scheduler(app, files_dir, config['SCHEDULER_INTERVAL_MINUTES'], 

190 first_run_at=datetime.now(timezone.utc)) 

191 

192 source = KafkaSource(config['KAFKA_BOOTSTRAP_SERVERS'], config['REPORTING_CONSUMER_GROUP'], 

193 topics, on_assign=stores.refresh) 

194 try: 

195 run_consumer_loop(source, stores, stop) 

196 finally: 

197 if scheduler is not None: 

198 scheduler.shutdown(wait=True) 

199 

200 

201if __name__ == '__main__': # pragma: no cover - the container entry point 

202 main()
