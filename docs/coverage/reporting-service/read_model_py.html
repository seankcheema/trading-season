1""" 

2Keeps the read model (models.py) in step with the files the consumer writes. 

3 

4The service has no database to ask. Each process holds the tables in an 

5in-memory SQLite database and fills them itself: sync() reads whatever has 

6been appended to the event files since it last looked and applies it 

7(projection.py). The first call reads everything; later calls read only the 

8new lines, so a request costs a few stat() calls when nothing has happened. 

9 

10The platform-records files are read before the trade-events files. Both 

11orders converge on the same rows, but reading the customers and accounts 

12first means an order never waits a round for the account it belongs to. 

13 

14One connection serves every thread of the process, so everything that 

15touches the tables happens while holding lock: syncing, querying, and 

16releasing the session afterwards. Reads are short and the console is polled 

17every few seconds per open screen, so one at a time is plenty. 

18 

19Nothing here is the only copy of anything. The files are the source, and if 

20one is replaced or removed the tables are emptied and rebuilt from what is 

21on disk. 

22""" 

23 

24import json 

25import logging 

26import threading 

27from decimal import Decimal 

28from pathlib import Path 

29 

30from event_store import PLATFORM_RECORDS, TRADE_EVENTS 

31from models import db 

32from projection import Unusable, apply_record, apply_trade_event 

33 

34logger = logging.getLogger(__name__) 

35 

36# Applied in this order on every sync: records, then order events. 

37SOURCES = ((PLATFORM_RECORDS, apply_record), (TRADE_EVENTS, apply_trade_event)) 

38 

39 

40class ReadModel: 

41 """The tail position of every event file, and the lock the tables are used under.""" 

42 

43 def __init__(self): 

44 self.lock = threading.RLock() 

45 self._positions = {} 

46 

47 def sync(self, session, files_dir): 

48 """ 

49 Apply every complete line appended since the last call and commit. 

50 Returns how many messages were applied. Hold lock while calling, and 

51 until the session is released. 

52 """ 

53 events_dir = Path(files_dir) / 'events' 

54 db.metadata.create_all(session.get_bind()) 

55 

56 sources = [(path, apply) for topic, apply in SOURCES 

57 for path in sorted(events_dir.glob(f'{topic}-p*.jsonl'))] 

58 if self._was_replaced(events_dir, [path for path, _ in sources]): 

59 logger.warning('Event files under %s were replaced; rebuilding the read model', events_dir) 

60 self._empty(session, events_dir) 

61 

62 applied = 0 

63 for path, apply in sources: 

64 applied += self._tail(session, path, apply) 

65 session.commit() 

66 if applied: 

67 logger.info('Read model applied %s message(s) from %s', applied, events_dir) 

68 return applied 

69 

70 def _was_replaced(self, events_dir, paths): 

71 """True when a file we have read from is gone or shorter than what we read of it.""" 

72 on_disk = {str(path): path.stat().st_size for path in paths} 

73 for known, position in self._positions.items(): 

74 if Path(known).parent != events_dir: 

75 continue 

76 if known not in on_disk or on_disk[known] < position: 

77 return True 

78 return False 

79 

80 def _empty(self, session, events_dir): 

81 for table in reversed(db.metadata.sorted_tables): 

82 session.execute(table.delete()) 

83 for known in [known for known in self._positions if Path(known).parent == events_dir]: 

84 del self._positions[known] 

85 

86 def _tail(self, session, path, apply): 

87 position = self._positions.get(str(path), 0) 

88 with open(path, 'rb') as handle: 

89 handle.seek(position) 

90 chunk = handle.read() 

91 # The consumer may be mid-write: a line counts once its newline is on disk. 

92 end = chunk.rfind(b'\n') 

93 if end < 0: 

94 return 0 

95 applied = 0 

96 for raw in chunk[:end].split(b'\n'): 

97 if not raw.strip(): 

98 continue 

99 try: 

100 # Money and quantities stay exact instead of passing through a float. 

101 message = json.loads(raw, parse_float=Decimal) 

102 except (ValueError, UnicodeDecodeError): 

103 logger.warning('Skipping unreadable line in %s', path.name) 

104 continue 

105 if not isinstance(message, dict): 

106 logger.warning('Skipping malformed line in %s', path.name) 

107 continue 

108 try: 

109 applied += 1 if apply(session, message) else 0 

110 except Unusable as reason: 

111 logger.warning('Skipping %s message partition=%s offset=%s: %s', 

112 path.name, message.get('partition'), message.get('offset'), reason) 

113 self._positions[str(path)] = position + end + 1 

114 return applied 

115
